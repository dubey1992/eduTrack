# The Python backend, pinned. Versions matter more than usual here: this has to
# reproduce an API the Flutter apps already depend on, and a silent upgrade that
# changes how DRF renders an error would be a contract break nobody asked for.

Django==5.1.4
djangorestframework==3.15.2

# psycopg 3, binary build - no local libpq or compiler needed, which matters on
# Windows and on shared hosting alike.
psycopg[binary]==3.2.3

# Test data, the equivalent of Laravel's factories.
factory-boy==3.3.1

# Passwords, in Laravel's format - see school/hashing.py. Both backends run
# against the same users table while the rollback window is open, so a password
# set by one has to be readable by the other.
bcrypt==4.2.1

# CORS. The Flutter web app is served from a different origin than the API, so
# the browser preflights every request; Laravel answers those through its own
# config/cors.php and this is the same allowance. Hand-rolling it was the
# alternative, and a preflight handled almost correctly is a security bug.
django-cors-headers==4.6.0

# The PDF receipt. Pure Python on purpose: shared hosting has no cairo or
# pango, which rules out WeasyPrint. Like dompdf on the Laravel side it
# understands only simple CSS, so the receipt template came across unchanged.
xhtml2pdf==0.2.16

# Provider credentials and the SMTP password are stored encrypted
# (docs/communication.md). Fernet - AES with an HMAC - from the one widely
# maintained Python library that ships it; the standard library has HMAC but
# no AES, and a hand-rolled cipher is exactly the code nobody should write.
# Already here transitively through xhtml2pdf; pinned now that it is used directly.
cryptography==50.0.1
